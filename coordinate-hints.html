getXCoordinates()
static double[] getXCoordinates() {
    double[] x = new double[workouts.size()];

    for (int i = 0; i < workouts.size(); i++) {
        x[i] = workouts.get(i).workoutNumber;
    }

    return x;
}
getYCoordinates()
static double[] getYCoordinates() {
    double[] y = new double[workouts.size()];

    for (int i = 0; i < workouts.size(); i++) {
        y[i] = workouts.get(i).getVolume();
    }

    return y;
}
